# Only needed for LLM_ENGINE=agent (coach_agent.py) — the default "direct"
# engine (coach.py + llm.py) doesn't import any of this, so it stays out of
# the base requirements.txt the same way the openai/gemini SDKs do.
#
#   pip install -r requirements-agent.txt
#   python -m rag.build_index      # one-time — builds rag/chroma_db/
#   LLM_ENGINE=agent python server.py
#
# Note: langchain-anthropic pulls in a newer `anthropic` SDK than the
# requirements.txt pin (0.119.0 -> 0.120.2 as of this writing). llm.py's
# direct-engine usage only calls the stable client.messages.create() API and
# has been tested against the newer version with no changes needed, but if
# you install both requirement files into the same environment, `anthropic`
# will end up at whatever langchain-anthropic resolves to, not the pin below.
langchain==1.3.14
langchain-anthropic==1.5.3

# Pinned explicitly even though langchain-anthropic already pulls it in
# transitively — coach_agent.py's tracing_enabled() and the LangSmith setup
# in its module docstring depend on this being present, so it's worth
# naming rather than leaving implicit. No account/API key ships with this;
# see coach_agent.py's docstring for the (optional) LANGSMITH_* env vars.
langsmith==0.10.15

# RAG (search_research_notes tool, rag/): fastembed is a local, ONNX-runtime
# embedding model (no torch, no API key — see rag/embeddings.py's docstring
# for why local embeddings were the deliberate choice, not just a convenient
# one). langchain-chroma is the vector-store client; chromadb is its engine.
fastembed==0.8.0
chromadb==1.5.9
langchain-chroma==1.1.0
langchain-text-splitters==1.1.2
